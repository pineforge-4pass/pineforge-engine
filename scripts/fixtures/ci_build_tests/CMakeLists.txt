cmake_minimum_required(VERSION 3.16)
project(RegisteredBuildControl LANGUAGES C CXX)
enable_testing()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_library(pineforge STATIC support.cpp)
add_library(pineforge_kernel STATIC support.cpp)
add_library(control_object OBJECT support.cpp)
add_executable(control_smoke smoke.cpp)
add_test(NAME control_smoke COMMAND control_smoke)
set_tests_properties(control_smoke PROPERTIES LABELS fast)
add_custom_target(pineforge_pr_smoke DEPENDS control_smoke)

# This ordinary Release target is registered but outside the smoke closure.
add_executable(control_deferred deferred.cpp)
add_test(NAME control_registered COMMAND control_deferred)
add_test(NAME control_alias COMMAND control_deferred)
if(PINEFORGE_BUILD_EXAMPLES)
    add_executable(control_example smoke.cpp)
    add_test(NAME control_example COMMAND control_example)
endif()
if(PINEFORGE_BUILD_TUTORIAL)
    add_library(control_tutorial MODULE support.cpp)
endif()
